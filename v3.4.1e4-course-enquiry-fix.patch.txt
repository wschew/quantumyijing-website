   OR rename it so it is not submitted as `website`.

   The normal registration form may keep its existing stable honeypot logic.

2) In the enquiry modal submit handler, build the request body exactly like this
   and POST it to the existing /api/enquiry endpoint:

const body = {
  name: fd.get('name'),
  email: fd.get('email'),
  phone: fd.get('phone'),
  country: fd.get('country'),

  // Reuse the already-tested CRM enquiry category.
  interest: 'Academy Course',

  // The enquiry modal may use "question" or "message" as its textarea name.
  message: String(fd.get('question') || fd.get('message') || '').trim(),

  consent: fd.get('consent'),

  // IMPORTANT v3.4.1e4:
  // Do NOT send a second enquiry-modal honeypot as `website`.
  // The stable /api/enquiry endpoint treats a non-empty website field as a bot
  // and intentionally returns {ok:true} without CRM insert or email.
  website: '',

  startedAt: Number(fd.get('startedAt')),
  language: root.dataset.lang,

  marketingSource: source,
  campaignCode: q.get('utm_campaign') || q.get('campaign') || '',
  landingPage: location.pathname,
  referrer: document.referrer,
  utmSource: q.get('utm_source') || '',
  utmMedium: q.get('utm_medium') || '',
  utmCampaign: q.get('utm_campaign') || '',
  utmContent: q.get('utm_content') || '',
  utmTerm: q.get('utm_term') || '',
  affiliateCode: q.get('aff') || q.get('affiliate') || '',

  // Preserve product attribution so CRM knows which course was enquired about.
  productId: P.id,
  productSlug: P.slug,

  // Enquiry only: DO NOT create an order.
  createOrder: false
};

const r = await fetch('/api/enquiry', {
  method: 'POST',
  headers: {'content-type': 'application/json'},
  body: JSON.stringify(body)
});

const d = await r.json();

if (!r.ok) {
  throw new Error(d.error || 'Unable to send enquiry.');
}

// Recommended success message:
m.textContent =
  root.dataset.lang === 'zh'
    ? '课程咨询已收到。参考编号：' + d.reference + '。我们会尽快回复您。'
    : 'Course enquiry received. Reference: ' + d.reference + '. We will reply as soon as possible.';

m.className = 'product-message success';
